checker = Path('tools/ci/check_town_movein_schedule_source.py')

old = '''    teleport_home = extract_method(npc, "private void AI_007_TownEntities_TeleportToHome(int homeFloorX, int homeFloorY)")

        "position.X",

        "homeFloorY",
        "height",
        "0.1f"
    ), "AI_007 home teleport bottom anchor")
'''

    require_ordered(teleport_home, (
        "homeFloorX",

        "AI_007_TryForcingSitting(homeFloorX, homeFloorY)"
